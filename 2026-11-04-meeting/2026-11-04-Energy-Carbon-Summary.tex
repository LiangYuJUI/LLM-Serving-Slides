\begin{frame}[t]{Cache Your Prompt When It's Green ---\\Carbon-Aware Caching for Large Language Model Serving}
\meta{Carbon-Aware KV Cache Management for LLM Serving}
\point{Motivation}{SSD-based prefix/KV caching avoids redundant prefill and operational carbon from electricity, but adds embodied carbon from SSD manufacturing.}
\point{Key Insight}{Carbon-efficient cache capacity depends on workload intensity and grid carbon intensity.}
\point{Method}{An ILP allocates SSD capacity dynamically. Least Carbon Savings (LCS) selects cache entries for replacement.}
\point{Experiments}{Llama-3 70B (INT8), $4\times$ L40 GPUs, LMCache, up to 16 TB SSD. Workloads: ShareGPT and TriviaQA.}
\point{Results}{In France, 15.1\% average carbon reduction vs.\ Full Cache, peaking at 25.3\% in multi-turn serving (70B). Reported latency SLO attainment: $>90\%$.}
\point{Conclusion}{Carbon-efficient caching balances avoided GPU computation against storage-related carbon costs.}
\src{Tian et al., \textit{Cache Your Prompt When It's Green --- Carbon-Aware Caching for Large Language Model Serving}.\\
SIGMETRICS 2026 / POMACS 10(1), Article 5.\quad
\href{https://doi.org/10.1145/3788087}{DOI: 10.1145/3788087}\quad
\href{https://arxiv.org/pdf/2505.23970v3}{Paper PDF}\quad
\href{https://greencache.persistentmemory.org}{Project}}
\end{frame}

\begin{frame}[t]{Beyond Energy: When Sustainability Dimensions\\Reshape LLM Serving Decisions}
\meta{Beyond Energy: Sustainability-Aware LLM Serving}
\point{Motivation}{Across deployment choices, minimizing energy need not minimize carbon, water, or biodiversity impacts.}
\point{Key Insight}{For fixed workload, location and time, operational-only accounting preserves the energy-based computing-configuration ranking in all four dimensions.}
\point{Method}{PRISM evaluates four environmental dimensions and routes across regions to minimize the largest relative penalty (regret) vs.\ each dimension's optimum.}
\point{Experiments}{Compare GPU/model configurations and regional deployment decisions using environmental-intensity data.}
\point{Results}{50.2\% median reduction in worst-case regret vs.\ the offline WaterWise adaptation in the 12-region evaluation (20 capacity seeds).}
\point{Conclusion}{Energy guides fixed-site operational optimization. Across regions/time, or with embodied impacts, rankings can differ.}
\src{Shi et al., \textit{Beyond Energy: When Sustainability Dimensions Reshape LLM Serving Decisions}.\\
2026.\quad \href{https://arxiv.org/abs/2609.35569}{arXiv:2609.35569}\quad
\href{https://arxiv.org/pdf/2609.35569}{Paper PDF}}
\end{frame}
